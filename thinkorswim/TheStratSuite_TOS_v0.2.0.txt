#
# Classification follows grammar/SPEC.md exactly:
#   - equal is not a break (high == prior high is not a 2u)
#   - above open is close > open; a close at the open buckets down
#   - only 2u/2d can fail; Reclaim is the default Failing 2 method
#
# Repaint: every painted historical bar uses only values that were fixed at that bar.
# Strat Candles grade the chart bar against the prior chart bar. FTFC Candles grade the chart
# bar's close against each higher timeframe's OPEN (fixed when that period starts), the same
# rule as the Suite's BARCOLOR-1. On stock charts with extended hours on, which daily candle a
# pre-market bar belongs to is still unconfirmed (probe P6). The strip and labels read the
# forming candles on the last bar.

declare upper;

# ============================================================================
# CLASSIFIER (grammar/SPEC.md sections 2-4)
# ============================================================================
# State codes: 0 no data, 1 1u, 2 1d, 3 2u, 4 2d, 5 F2u, 6 F2d, 7 3u, 8 3d
# method: 0 Reclaim, 1 Open, 2 Reclaim + Open, 3 Reclaim OR Open
script StratState {
    input prevH = 0.0;
    input prevL = 0.0;
    input o = 0.0;
    input h = 0.0;
    input l = 0.0;
    input c = 0.0;
    input method = 0;
    input detectF2 = yes;
    def valid = !IsNaN(prevH) and !IsNaN(prevL) and !IsNaN(o) and !IsNaN(h) and !IsNaN(l) and !IsNaN(c) and h >= l and prevH >= prevL;
    def brokeHigh = h > prevH;
    def brokeLow = l < prevL;
    def aboveOpen = c > o;
    def reclaimed = c <= prevH and c >= prevL;
    def f2u = detectF2 and (if method == 0 then reclaimed else if method == 1 then !aboveOpen else if method == 2 then reclaimed and !aboveOpen else reclaimed or !aboveOpen);
    def f2d = detectF2 and (if method == 0 then reclaimed else if method == 1 then aboveOpen else if method == 2 then reclaimed and aboveOpen else reclaimed or aboveOpen);
    plot state = if !valid then 0
        else if brokeHigh and brokeLow then (if aboveOpen then 7 else 8)
        else if brokeHigh then (if f2u then 5 else 3)
        else if brokeLow then (if f2d then 6 else 4)
        else (if aboveOpen then 1 else 2);
}

# the chart's own timeframe for data (aN) and is then left out, like the Suite's isLowerTF cells.

# ============================================================================
# FULL TIMEFRAME CONTINUITY (grammar/SPEC.md section 7)
# ============================================================================
# FTFC Candles grade each chart bar's close against each monitored timeframe's open (the
# history rule, as in the Suite's bar paint). The FTFC label grades each timeframe's own served
# close, as the Suite's FTFC row does, so it agrees with the strip. The two can differ on the
# last bar when a served close is not the chart close: a futures settlement close, or an
# RTH-only daily close on a stock chart with extended hours on. A close at the open is down.
def ftfcAny = use1 or use2 or use3 or use4 or use5 or use6;
def ftfcUp = ftfcAny and (!use1 or close > o1) and (!use2 or close > o2) and (!use3 or close > o3) and (!use4 or close > o4) and (!use5 or close > o5) and (!use6 or close > o6);
def ftfcDown = ftfcAny and (!use1 or close <= o1) and (!use2 or close <= o2) and (!use3 or close <= o3) and (!use4 or close <= o4) and (!use5 or close <= o5) and (!use6 or close <= o6);
def lblUse1 = use1 and !IsNaN(c1);
def lblUse2 = use2 and !IsNaN(c2);
def lblUse3 = use3 and !IsNaN(c3);
def lblUse4 = use4 and !IsNaN(c4);
def lblUse5 = use5 and !IsNaN(c5);
def lblUse6 = use6 and !IsNaN(c6);
def lblAny = lblUse1 or lblUse2 or lblUse3 or lblUse4 or lblUse5 or lblUse6;
def lblUp = lblAny and (!lblUse1 or c1 > o1) and (!lblUse2 or c2 > o2) and (!lblUse3 or c3 > o3) and (!lblUse4 or c4 > o4) and (!lblUse5 or c5 > o5) and (!lblUse6 or c6 > o6);
def lblDown = lblAny and (!lblUse1 or c1 <= o1) and (!lblUse2 or c2 <= o2) and (!lblUse3 or c3 <= o3) and (!lblUse4 or c4 <= o4) and (!lblUse5 or c5 <= o5) and (!lblUse6 or c6 <= o6);

        else if chartState == 7 and paintOutsideBars then GlobalColor("3u")
        else if chartState == 8 and paintOutsideBars then GlobalColor("3d")
        else Color.CURRENT)
    else if ftfcMode and ftfcAny then (
        if ftfcUp then GlobalColor("FTFC Up")
        else if ftfcDown then GlobalColor("FTFC Down")
        else if paintFtfcConflict then GlobalColor("FTFC Conflict")
        else Color.CURRENT)
    else Color.CURRENT);
