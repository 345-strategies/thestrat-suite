# Optional, and separate from the study itself. The full port depends on a few answers that
# the thinkScript manual does not settle (thinkorswim/FULL_PORT_PLAN.md, "Open questions to
# settle in the app"). This study draws labels and one chart bubble, and fires one alert with
# no sound only if the tester turns fireP10Alert on. It never draws signals or places orders.
# Add it next to TheStratSuite as a second study, follow TESTER_GUIDE.md Test 4, and send
# screenshots. P-numbers match the plan.
#
# Run it on intraday or daily charts only: it reads D, W, M, 3M and 12M data, and
# thinkorswim refuses a timeframe below the chart's.

declare upper;

input showP1 = yes;
input showP4 = yes;
input showP6 = yes;
input showP10 = yes;
input fireP10Alert = no;
input showP22 = yes;

# ============================================================================
# P1: what [1] means on a def that holds higher-timeframe data
# ============================================================================
# Each offset is taken in its own def, so no label mixes an offset with chart data.
#   direct  = close(period = DAY)[1], the usual "prior day close" idiom
#   def     = a pure higher-timeframe def, then [1]
#   mixed   = a def mixed with chart data (+ 0 * close), then [1]
#   mirror  = a def gated by a date function (GetYYYYMMDD), then [1]
#   gatedDef = a def gated by IsNaN(close), then [1]
#   inline  = the offset taken inside the IsNaN(close) gate
#   defInMixed = the pure def's [1] used inside a mixed expression
# If a field equals direct, offsets in that form step whole days. If it equals D[0] (today's
# D close), they step chart bars. "same D" says whether the prior chart bar is in today's D;
# if it reads no, the chart is on a session's first bar and the answer is not readable yet.
def dClose = close(period = AggregationPeriod.DAY);
def dMixed = dClose + 0 * close;
def dMirror = if GetYYYYMMDD() > 0 then close(period = AggregationPeriod.DAY) else 0;
def dGated = if IsNaN(close) then 0 else close(period = AggregationPeriod.DAY);
def tDirect = close(period = AggregationPeriod.DAY)[1];
def tDef = dClose[1];
def tMixed = dMixed[1];
def tMirror = dMirror[1];
def tGated = dGated[1];
def tInline = if IsNaN(close) then 0 else close(period = AggregationPeriod.DAY)[1];
def tDefInMixed = dClose[1] + 0 * close;
def sameD = GetYYYYMMDD() == GetYYYYMMDD()[1];
AddLabel(showP1,
    "P1a " + GetSymbol() + " direct " + AsPrice(tDirect) + " | def " + AsPrice(tDef)
    + " | mixed " + AsPrice(tMixed) + " | mirror " + AsPrice(tMirror),
    Color.LIGHT_GRAY);
AddLabel(showP1,
    "P1b gatedDef " + AsPrice(tGated) + " | inline " + AsPrice(tInline)
    + " | defInMixed " + AsPrice(tDefInMixed),
    Color.LIGHT_GRAY);
AddLabel(showP1,
    "P1c D[0] " + AsPrice(dClose) + " | last " + AsPrice(close) + " | same D " + (if sameD then "yes" else "no"),
    Color.LIGHT_GRAY);

# ============================================================================
# P4: how far back higher-timeframe data reaches on this chart
# ============================================================================
# Exhaustion levels need 52 candles of a timeframe. "52+" means the 52nd candle back exists.
def d52 = !IsNaN(high(period = AggregationPeriod.DAY)[51]);
def d5 = !IsNaN(high(period = AggregationPeriod.DAY)[4]);
def w52 = !IsNaN(high(period = AggregationPeriod.WEEK)[51]);
def w5 = !IsNaN(high(period = AggregationPeriod.WEEK)[4]);
def m52 = !IsNaN(high(period = AggregationPeriod.MONTH)[51]);
def m5 = !IsNaN(high(period = AggregationPeriod.MONTH)[4]);
def q52 = !IsNaN(high(period = AggregationPeriod.QUARTER)[51]);
def q5 = !IsNaN(high(period = AggregationPeriod.QUARTER)[4]);
def y52 = !IsNaN(high(period = AggregationPeriod.YEAR)[51]);
def y5 = !IsNaN(high(period = AggregationPeriod.YEAR)[4]);
AddLabel(showP4,
    "P4 history: chart bars " + BarNumber()
    + " | D " + (if d52 then "52+" else if d5 then "5-51" else "under 5")
    + " | W " + (if w52 then "52+" else if w5 then "5-51" else "under 5")
    + " | M " + (if m52 then "52+" else if m5 then "5-51" else "under 5")
    + " | 3M " + (if q52 then "52+" else if q5 then "5-51" else "under 5")
    + " | 12M " + (if y52 then "52+" else if y5 then "5-51" else "under 5"),
    Color.LIGHT_GRAY);

# ============================================================================
# P6: which daily candle a pre-market bar belongs to (stocks, extended hours on)
# ============================================================================
# Read on an intraday stock chart with extended hours on. Everything is recorded at the most
# recent bar that starts at 9:30 ET (6:30 AM PT), so all values describe one morning whenever
# the screenshot is taken.
#   7:00 D open = the D open as seen from the bar starting at 7:00 ET (4 AM PT)
#   9:30 D open = the D open as seen from the 9:30 ET bar
#   9:30 bar / first bar = the opens of those chart bars
#   prior 9:30 D open = the previous day's 9:30 D open
# 7:00 D open equal to 9:30 D open: pre-market bars belong to today's D.
# 7:00 D open equal to prior 9:30 D open: they belong to yesterday's D.
# 9:30 D open equal to the 9:30 bar's open: D does not start in the pre-market. Equal to the
# first bar's open: D starts in the pre-market. Where D ends is P6b.
# "no 7:00 bar" means no bar starts at 7:00 ET that morning. N/A in a D open field means the
# bar exists but open(period = DAY) is NaN on it, which is an answer too: that bar maps to no D.
# If the whole label reads N/A, no bar starts at 9:30 on this chart.
def intradayP6 = GetAggregationPeriod() < AggregationPeriod.DAY;
def at7 = intradayP6 and SecondsFromTime(0700) == 0;
def at930 = intradayP6 and SecondsFromTime(0930) == 0;
def newDate = GetYYYYMMDD() != GetYYYYMMDD()[1];
def firstOpen = CompoundValue(1, if newDate then open else firstOpen[1], Double.NaN);
def dAt7 = CompoundValue(1, if at7 then open(period = AggregationPeriod.DAY) else if newDate then Double.NaN else dAt7[1], Double.NaN);
def had7 = CompoundValue(1, if at7 then 1 else if newDate then 0 else had7[1], 0);
def snapDate = CompoundValue(1, if at930 then GetYYYYMMDD() else snapDate[1], Double.NaN);
def snap7 = CompoundValue(1, if at930 then dAt7 else snap7[1], Double.NaN);
def snapHad7 = CompoundValue(1, if at930 then had7 else snapHad7[1], 0);
def snapD = CompoundValue(1, if at930 then open(period = AggregationPeriod.DAY) else snapD[1], Double.NaN);
def snapBar = CompoundValue(1, if at930 then open else snapBar[1], Double.NaN);
def snapFirst = CompoundValue(1, if at930 then firstOpen else snapFirst[1], Double.NaN);
def snapPrevD = CompoundValue(1, if at930 then snapD[1] else snapPrevD[1], Double.NaN);
AddLabel(showP6,
    "P6 morning of " + AsPrice(snapDate) + ": 7:00 D open " + (if snapHad7 then AsPrice(snap7) else "no 7:00 bar")
    + " | 9:30 D open " + AsPrice(snapD)
    + " | 9:30 bar " + AsPrice(snapBar) + " | first bar " + AsPrice(snapFirst)
    + " | prior 9:30 D open " + AsPrice(snapPrevD),
    Color.LIGHT_GRAY);

# P6b: where D ends. Read it on the same chart after 4:05 PM ET (1:05 PM PT) and before
# midnight ET, so the last bar is an after-hours bar of the same date as the morning above.
#   D now = the D candle the last bar belongs to
#   RTH = today's 9:30-16:00 ET chart bars; after-hours = today's chart bars from 16:00 ET
# D now open equal to the 9:30 D open above: the last bar is in today's D. Then D now high,
# low and close equal to RTH (close can differ by the official closing print) mean D ends at
# 16:00; D now close equal to last, or high/low reaching the after-hours high/low, mean D takes
# after-hours bars. D now open equal to the after-hours open: after-hours bars start a new D.
# N/A for D now: after-hours bars map to no D.
def dOpenNow = open(period = AggregationPeriod.DAY);
def dHighNow = high(period = AggregationPeriod.DAY);
def dLowNow = low(period = AggregationPeriod.DAY);
def rthBar = intradayP6 and SecondsFromTime(0930) >= 0 and SecondsTillTime(1600) > 0;
def ahBar = intradayP6 and SecondsFromTime(1600) >= 0;
def rthH = CompoundValue(1, if rthBar then (if newDate or IsNaN(rthH[1]) then high else Max(rthH[1], high)) else if newDate then Double.NaN else rthH[1], Double.NaN);
def rthL = CompoundValue(1, if rthBar then (if newDate or IsNaN(rthL[1]) then low else Min(rthL[1], low)) else if newDate then Double.NaN else rthL[1], Double.NaN);
def rthC = CompoundValue(1, if rthBar then close else if newDate then Double.NaN else rthC[1], Double.NaN);
def ahO = CompoundValue(1, if ahBar and (newDate or IsNaN(ahO[1])) then open else if newDate then Double.NaN else ahO[1], Double.NaN);
def ahH = CompoundValue(1, if ahBar then (if newDate or IsNaN(ahH[1]) then high else Max(ahH[1], high)) else if newDate then Double.NaN else ahH[1], Double.NaN);
def ahL = CompoundValue(1, if ahBar then (if newDate or IsNaN(ahL[1]) then low else Min(ahL[1], low)) else if newDate then Double.NaN else ahL[1], Double.NaN);
AddLabel(showP6 and intradayP6,
    "P6b D now O/H/L/C " + AsPrice(dOpenNow) + "/" + AsPrice(dHighNow) + "/" + AsPrice(dLowNow) + "/" + AsPrice(dClose)
    + " | RTH H/L/C " + AsPrice(rthH) + "/" + AsPrice(rthL) + "/" + AsPrice(rthC)
    + " | after-hours O/H/L " + AsPrice(ahO) + "/" + AsPrice(ahH) + "/" + AsPrice(ahL)
    + " | last " + AsPrice(close),
    Color.LIGHT_GRAY);

# ============================================================================
# P10: which symbols render where the Suite uses them
# ============================================================================
# The Suite's chart labels use arrows and the diamond (bubbles in the full port). Its alert
# text adds the two colored circles. Each surface is tested once.
AddLabel(showP10, "P10 label: ↑ ↓ ◆ 🟢 🔴", Color.LIGHT_GRAY);
AddChartBubble(showP10 and !IsNaN(close) and IsNaN(close[-1]), high, "P10 bubble: ↑ ↓ ◆ 🟢 🔴", Color.LIGHT_GRAY, yes);
Alert(fireP10Alert, "P10 alert: ↑ ↓ ◆ 🟢 🔴", Alert.ONCE, Sound.NoSound);
