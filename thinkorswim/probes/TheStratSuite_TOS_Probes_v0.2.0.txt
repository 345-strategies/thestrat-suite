# Optional, and separate from the study itself. The full port depends on a few answers that
# the thinkScript manual does not settle (thinkorswim/FULL_PORT_PLAN.md, "Open questions to
# settle in the app"). This study draws gray and colored labels only, never signals or orders.
# Add it next to TheStratSuite as a second study, follow TESTER_GUIDE.md Test 4, and send a
# screenshot of its labels. P-numbers match the plan.
#
# Run it on intraday or daily charts only: it reads D, W, M, 3M and 12M data, and
# thinkorswim refuses a timeframe below the chart's.

declare upper;

input showP1 = yes;
input showP4 = yes;
input showP6 = yes;
input showP10 = yes;
input showP22 = yes;

# ============================================================================
# P1: what [1] means on a def that holds higher-timeframe data
# ============================================================================
# direct[1] is the usual "prior day close" idiom. If def[1] matches it, offsets on stored
# higher-timeframe values step whole days. If def[1] matches "last" instead, they step chart bars.
def dClose = close(period = AggregationPeriod.DAY);
def dMixed = dClose + 0 * close;
def dGated = if IsNaN(close) then 0 else close(period = AggregationPeriod.DAY);
AddLabel(showP1,
    "P1 " + GetSymbol() + " D close: direct[1] " + AsPrice(close(period = AggregationPeriod.DAY)[1])
    + " | def[1] " + AsPrice(dClose[1]) + " | mixed[1] " + AsPrice(dMixed[1])
    + " | gated[1] " + AsPrice(dGated[1]) + " | last " + AsPrice(close),
    Color.LIGHT_GRAY);

# ============================================================================
# P4: how far back higher-timeframe data reaches on this chart
# ============================================================================
# Exhaustion levels need 52 candles of a timeframe. "52+" means the 52nd candle back exists.
def w52 = !IsNaN(high(period = AggregationPeriod.WEEK)[51]);
def w5 = !IsNaN(high(period = AggregationPeriod.WEEK)[4]);
def m52 = !IsNaN(high(period = AggregationPeriod.MONTH)[51]);
def m5 = !IsNaN(high(period = AggregationPeriod.MONTH)[4]);
def q52 = !IsNaN(high(period = AggregationPeriod.QUARTER)[51]);
def q5 = !IsNaN(high(period = AggregationPeriod.QUARTER)[4]);
def y52 = !IsNaN(high(period = AggregationPeriod.YEAR)[51]);
def y5 = !IsNaN(high(period = AggregationPeriod.YEAR)[4]);
AddLabel(showP4,
    "P4 history: chart bars " + BarNumber()
    + " | W " + (if w52 then "52+" else if w5 then "5-51" else "under 5")
    + " | M " + (if m52 then "52+" else if m5 then "5-51" else "under 5")
    + " | 3M " + (if q52 then "52+" else if q5 then "5-51" else "under 5")
    + " | 12M " + (if y52 then "52+" else if y5 then "5-51" else "under 5"),
    Color.LIGHT_GRAY);

# ============================================================================
# P6: which daily candle a pre-market bar belongs to (stocks, extended hours on)
# ============================================================================
# Records the D open as seen from the 7:00 ET bar (4 AM PT), next to the 9:30 ET open of the
# same day and of the day before. Shows N/A on a chart without extended hours.
def sft7 = SecondsFromTime(0700);
def at7 = sft7 >= 0 and sft7[1] < 0;
def dOpenAt7 = CompoundValue(1, if at7 then open(period = AggregationPeriod.DAY) else dOpenAt7[1], Double.NaN);
def sft930 = SecondsFromTime(0930);
def at930 = sft930 >= 0 and sft930[1] < 0;
def rthOpen = CompoundValue(1, if at930 then open else rthOpen[1], Double.NaN);
def prevRthOpen = CompoundValue(1, if at930 then rthOpen[1] else prevRthOpen[1], Double.NaN);
AddLabel(showP6,
    "P6 D open seen at 7:00 ET " + AsPrice(dOpenAt7) + " | 9:30 ET open " + AsPrice(rthOpen)
    + " | prior 9:30 ET open " + AsPrice(prevRthOpen),
    Color.LIGHT_GRAY);

# ============================================================================
# P10: which symbols render (the Suite's labels and alerts use these)
# ============================================================================
AddLabel(showP10, "P10 glyphs: ↑ ↓ ◆ ♦ ✓ 🟢 🔴", Color.LIGHT_GRAY);

# ============================================================================
# P22: is the platform's label text readable on each Suite color?
# ============================================================================
AddLabel(showP22, "P22 1u", CreateColor(255, 235, 59));
AddLabel(showP22, "1d", CreateColor(255, 152, 0));
AddLabel(showP22, "2u", CreateColor(76, 175, 80));
AddLabel(showP22, "2d", CreateColor(242, 54, 69));
AddLabel(showP22, "F2u", CreateColor(247, 124, 128));
AddLabel(showP22, "F2d", CreateColor(129, 199, 132));
AddLabel(showP22, "3u", CreateColor(8, 153, 129));
AddLabel(showP22, "3d", CreateColor(233, 30, 99));
AddLabel(showP22, "Conflict", CreateColor(128, 128, 128));
